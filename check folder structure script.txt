$root = (Get-Location).Path
$output = Join-Path ([Environment]::GetFolderPath("Desktop")) "ecommerce_folder_structure.txt"

$exclude = @(".git", ".pytest_cache", "__pycache__", ".venv", "venv", "node_modules")
$lines = [System.Collections.Generic.List[string]]::new()
$lines.Add("PROJECT ROOT: $root")
$lines.Add("GENERATED: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')")
$lines.Add("")

function Get-FolderTree {
    param([string]$Path, [string]$Prefix)
    $items = @(Get-ChildItem -LiteralPath $Path -Force -ErrorAction SilentlyContinue | Where-Object { $_.Name -notin $exclude } | Sort-Object @{Expression={if ($_.PSIsContainer) {0} else {1}}}, Name)
    for ($i = 0; $i -lt $items.Count; $i++) {
        $item = $items[$i]
        $last = ($i -eq $items.Count - 1)
        $branch = if ($last) { "+-- " } else { "|-- " }
        $lines.Add("$Prefix$branch$($item.Name)")
        if ($item.PSIsContainer) {
            $nextPrefix = $Prefix + $(if ($last) { "    " } else { "|   " })
            Get-FolderTree -Path $item.FullName -Prefix $nextPrefix
        }
    }
}

Get-FolderTree -Path $root -Prefix ""
$lines | Set-Content -LiteralPath $output -Encoding UTF8
Write-Host "Inventory created: $output" -ForegroundColor Green
Write-Host "Project root: $root"
